cmake_minimum_required(VERSION 3.24)
project(TRPXeFGPrototype LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The XeFG prototype requires Windows D3D11/D3D12")
endif()
set(TRP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(TRP_DEPENDENCIES "${TRP_ROOT}/.dependencies")
if(DEFINED ENV{TRP_DEPENDENCY_ROOT})
    file(TO_CMAKE_PATH "$ENV{TRP_DEPENDENCY_ROOT}" TRP_DEPENDENCIES)
endif()
set(XESS_SDK "${TRP_DEPENDENCIES}/xess" CACHE PATH "Pinned Intel XeSS SDK root")
if(NOT EXISTS "${XESS_SDK}/inc/xess_fg/xefg_swapchain_d3d12.h")
    message(FATAL_ERROR "Set XESS_SDK to an Intel SDK containing the public XeFG and XeLL headers")
endif()
add_executable(TRPXeFGPrototype XeFGPrototype.cpp experimental/XeFGUnlock.cpp "${TRP_ROOT}/src/FrameGen/SourceDLSSGInterop.cpp")
target_compile_features(TRPXeFGPrototype PRIVATE cxx_std_23)
target_compile_definitions(TRPXeFGPrototype PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(TRPXeFGPrototype PRIVATE "${TRP_ROOT}/src" "${XESS_SDK}/inc")
target_link_libraries(TRPXeFGPrototype PRIVATE d3d11 d3d12 dxgi user32 bcrypt)
add_executable(TRPXeFGUnlockTests experimental/XeFGUnlockTests.cpp experimental/XeFGUnlock.cpp)
target_compile_features(TRPXeFGUnlockTests PRIVATE cxx_std_23)
target_compile_definitions(TRPXeFGUnlockTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(TRPXeFGUnlockTests PRIVATE bcrypt)
add_executable(TRPXeFGInteropTests "${TRP_ROOT}/tests/SourceDLSSGInteropFenceTests.cpp"
    "${TRP_ROOT}/src/FrameGen/SourceDLSSGInterop.cpp")
target_compile_features(TRPXeFGInteropTests PRIVATE cxx_std_23)
target_compile_definitions(TRPXeFGInteropTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(TRPXeFGInteropTests PRIVATE "${TRP_ROOT}/src")
target_link_libraries(TRPXeFGInteropTests PRIVATE d3d11 d3d12 dxgi)
if(MSVC)
    target_compile_options(TRPXeFGPrototype PRIVATE /W4 /permissive- /O2)
    target_compile_options(TRPXeFGInteropTests PRIVATE /W4 /permissive- /O2)
endif()
enable_testing()
add_test(NAME XeFGUnlockTransaction COMMAND TRPXeFGUnlockTests)
add_test(NAME XeFGProductionInterop COMMAND TRPXeFGInteropTests)
set_tests_properties(XeFGProductionInterop PROPERTIES TIMEOUT 60)
add_executable(TRPXeFGInputReuseTests XeFGInputReuseTests.cpp "${TRP_ROOT}/src/FrameGen/SourceDLSSGInterop.cpp")
target_compile_features(TRPXeFGInputReuseTests PRIVATE cxx_std_23)
target_compile_definitions(TRPXeFGInputReuseTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(TRPXeFGInputReuseTests PRIVATE "${TRP_ROOT}/src")
target_link_libraries(TRPXeFGInputReuseTests PRIVATE d3d11 d3d12 dxgi)
add_test(NAME XeFGInputReuse COMMAND TRPXeFGInputReuseTests)
set_tests_properties(XeFGInputReuse PROPERTIES TIMEOUT 30)
# SDK execution is explicit: no automatic DLL acquisition or game deployment.
set(STREAMLINE_INCLUDE "${TRP_DEPENDENCIES}/Streamline/include" CACHE PATH "Streamline public headers")
add_executable(TRPXeFGPresenterTests XeFGPresenterTests.cpp "${TRP_ROOT}/src/FrameGen/XeFGPresenter.cpp"
    "${TRP_ROOT}/src/FrameGen/SourceDLSSGInterop.cpp" "${TRP_ROOT}/src/FrameGen/SourceDLSSGSession.cpp")
target_compile_features(TRPXeFGPresenterTests PRIVATE cxx_std_23)
target_compile_definitions(TRPXeFGPresenterTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_include_directories(TRPXeFGPresenterTests PRIVATE "${TRP_ROOT}/src" "${XESS_SDK}/inc" "${STREAMLINE_INCLUDE}")
target_link_libraries(TRPXeFGPresenterTests PRIVATE d3d11 d3d12 dxgi user32 dbghelp)
add_executable(TRPXeSSUpscalerTests XeSSUpscalerTests.cpp "${TRP_ROOT}/src/FrameGen/XeSSUpscaler.cpp"
    "${TRP_ROOT}/src/FrameGen/SourceDLSSGInterop.cpp")
target_compile_features(TRPXeSSUpscalerTests PRIVATE cxx_std_23)
target_compile_definitions(TRPXeSSUpscalerTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_include_directories(TRPXeSSUpscalerTests PRIVATE "${TRP_ROOT}/src" "${XESS_SDK}/inc")
target_link_libraries(TRPXeSSUpscalerTests PRIVATE d3d11 d3d12 dxgi d3dcompiler user32)
